```bash
    "Woodpecker CI": ["ci", "cd", "devops"],
    "Pa11y": ["accessibility", "a11y", "testing", "inclusion"],

with open('data/services.yaml', 'r') as f:
    content = f.read()

# For each service, parse existing tags, append edu_specific, deduplicate, rewrite
# Use regex to find and replace each tags: [...] line
for name, extra_tags in edu_specific.items():
    escaped = re.escape(name)
    
    # Find the service block and extract its tags
    pattern = rf'(- name: "{escaped}".*?tags: \[)([^\]]*)(\])'
    match = re.search(pattern, content, re.DOTALL)
    if not match:
        print(f"✗ {name} — tags not found")
        continue
    
    prefix = match.group(1)
    existing_str = match.group(2)
    suffix = match.group(3)
    
    # Parse existing tags
    existing = [t.strip().strip('"') for t in existing_str.split(',') if t.strip()]
    
    # Merge new tags, preserving order and avoiding duplicates
    all_tags = list(existing)
    for t in extra_tags:
        if t not in all_tags:
            all_tags.append(t)
    
    # Rebuild tags line
    new_tags_str = ', '.join(f'"{t}"' for t in all_tags)
    replacement = f'{prefix}{new_tags_str}{suffix}'
    
    content = content[:match.start()] + replacement + content[match.end():]
    print(f"✓ {name} ({len(existing)} → {len(all_tags)} tags)")

with open('data/services.yaml', 'w') as f:
    f.write(content)

print("\nDone!")
```

### Task 5: Category gradient backgrounds (JS + CSS)

**Files:**
- Modify: `script.js` (generate gradient from `category.color`)
- Modify: `styles.css` (remove hardcoded gradient default)

- [ ] **Step 1: Update renderLandscape to use color-based gradient**

In `script.js`, find the `.category-header` HTML template and change the `style` attribute from using `border-left` only to also include a `background`:

Current line (around line 58):
```js
<div class="category-header" style="border-left: 4px solid ${category.color}">
```

Change to:
```js
<div class="category-header" style="border-left: 4px solid ${category.color}; background: linear-gradient(135deg, ${category.color}22 0%, ${category.color}44 100%)">
```

The `22` and `44` are 6-digit hex alpha values — but CSS hex alpha requires 8 digits. Actually, the correct approach is to use `rgba()` or to apply the gradient via an inline style on the element itself.

Better approach — put the base gradient in CSS, and set only the color via JS:

In styles.css, change:
```css
.category-header {
  background: linear-gradient(135deg, rgba(74, 144, 226, 0.2) 0%, rgba(80, 200, 120, 0.2) 100%);
  padding: 1.5rem 2rem;
  border-bottom: 1px solid rgba(255, 255, 255, 0.1);
}
```

To:
```css
.category-header {
  padding: 1.5rem 2rem;
  border-bottom: 1px solid rgba(255, 255, 255, 0.1);
}
```

Then in `script.js`, remove the hardcoded `style` attribute border-left and instead use JS to set both the border and background:

In `renderLandscape()`:
```js
categoryEl.innerHTML = `
  <div class="category-header" style="border-left: 4px solid ${category.color}">
    <h2>${category.icon} ${category.name} <span class="count-badge">${totalItems}</span></h2>
    <p>${category.description}</p>
  </div>
`;
```

Change to use a data attribute and CSS:

```js
categoryEl.innerHTML = `
  <div class="category-header" style="border-left: 4px solid ${category.color}; background: linear-gradient(135deg, ${category.color}22 0%, ${category.color}44 100%)">
    <h2>${category.icon} ${category.name} <span class="count-badge">${totalItems}</span></h2>
    <p>${category.description}</p>
  </div>
`;
```

Note: The `22` and `44` suffixes on a hex color only work in modern CSS (8-digit hex). To be safe, we'll use `rgba()` applied via JS color conversion. Alternative: apply the class `.category-header` normally and set the `style` attribute with both border and background. The current approach already uses inline styles — we're just adding the background alongside.

```js
  renderLandscape() {
    const container = document.getElementById('landscape');
    const existing = container.querySelectorAll('.category');
    
    if (existing.length > 0) {
      existing.forEach(el => el.classList.add('filtering'));
      setTimeout(() => {
        container.innerHTML = '';
        this._renderLandscapeContent(container);
      }, 200);
      return;
    }

    this._renderLandscapeContent(container);

  _renderLandscapeContent(container) {
    let hasVisible = false;
    this.data.categories.forEach(category => {
      const visibleSubcategories = category.subcategories
        .map(sub => ({
          ...sub,
          items: sub.items.filter(item => this.matchesFilter(item))
        }))
        .filter(sub => sub.items.length > 0);

      if (visibleSubcategories.length === 0) return;

      hasVisible = true;
      const categoryEl = document.createElement('div');
      categoryEl.className = 'category';
      const totalItems = category.subcategories.reduce((sum, s) => sum + (s.items || []).length, 0);
      categoryEl.innerHTML = `
        <div class="category-header" style="border-left: 4px solid ${category.color}; background: linear-gradient(135deg, ${category.color}22 0%, ${category.color}44 100%)">
          <h2>${category.icon} ${category.name} <span class="count-badge">${totalItems}</span></h2>
          <p>${category.description}</p>
        </div>
      `;

      visibleSubcategories.forEach(sub => {
        const subEl = document.createElement('div');
        subEl.className = 'subcategory';
        subEl.innerHTML = `
          <h3 class="subcategory-title">${sub.name}</h3>
          <div class="items-grid">
            ${sub.items.map(item => this.renderItem(item)).join('')}
          </div>
        `;
        categoryEl.appendChild(subEl);
      });

      container.appendChild(categoryEl);
    });

    if (!hasVisible) {
      container.innerHTML = `<div class="empty-state"> ... </div>`;
    }
  }
```
